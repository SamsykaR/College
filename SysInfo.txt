#!/bin/bash

report_uptime () {
		echo " 
			System Uptime: $(date +"%x %r %Z")
		"
		return
}

report_disk_space () {
		echo"
			Disk Space Utilization: $(df -h)
		"
		return
}

report_home_space () {
		if [[ $(id -u) -eq 0 ]]; then
			echo "
				Home Space Utilization (All Users): $(du -sh /home/*)
			"
		else
			echo "
				Home Space Utilization ($USER): $(du -sh $HOME)
			"
		fi
		return
}

report_main_info () {
		echo " 
			Core name: $(uname -a | cut -d' ' -f1)
			Conect name: $(uname -a |  cut -d ' ' -f2)
			Core version: $(uname -a |  cut -d ' ' -f3)
			Reliase date: $(uname -a |  cut -d ' ' -f4,5,6)
			Platform: $(uname -a |  cut -d ' ' -f7)
		"
		return
}

while true; do

echo "
			Menu
	1. Uptime
	2. Disk Space
	3. Home Space
	4. Main info 
	5. Clear output
	6. Exit
"

read -p "Your choise: "
if [[REPLY == 1]] report_uptime() fi
if [[REPLY == 2]] report_disk_space() fi
if [[REPLY == 3]] report_home_space() fi
if [[REPLY == 4]] report_main_info() fi
if [[REPLY == 5]] Clear fi
if [[REPLY == 6]] exit fi
done
